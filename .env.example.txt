# ══════════════════════════════════════════════════════════════════════════
# TW SelfSelect Trader — 環境設定範本
# 用法：cp .env.example.txt .env 之後填入自己的金鑰與名單。
# ⚠️ `.env` 永遠不進版控（`.gitignore` 已排除）；本檔只放範例值。
# ══════════════════════════════════════════════════════════════════════════

# ── 1. 資金 ─────────────────────────────────────────────────────────────
TOTAL_CAPITAL=500000            # 可動用總資金（NT$）。每檔買入上限 = 該檔 alloc% × 本值
TOTAL_CAPITAL_ORIGINAL=500000   # 原始投入資本（報告顯示用；不含後續滾入的獲利）
# 入金／提領紀錄放在 capital.txt（每行 `金額, YYYY/MM/DD`）。主程式每天檢查新增的列，
# 自動調整 TOTAL_CAPITAL；想手動同步就直接改上面兩行。

# ── 2. 券商 ─────────────────────────────────────────────────────────────
# esun = 玉山證券（行情＋下單）；kgi = 凱基（mock/實單）
BROKER=esun
USE_REAL_API=false              # BROKER=kgi 時才有意義；BROKER=esun 一律視為 true

# 玉山證券（需自備 esun_marketdata / esun_trade 的 whl 與憑證；見使用手冊「券商串接」）
ESUN_ENVIRONMENT=simulation     # simulation=模擬帳戶（預設）｜real=真錢，切換前必讀手冊
ESUN_ACCOUNT=your_account_id
ESUN_API_KEY=your_esun_api_key
ESUN_API_SECRET=your_esun_api_secret
ESUN_CERT_PATH=esun_sdk/your_cert.p12
ESUN_CERT_PASSWORD=your_esun_cert_password
ESUN_ACCOUNT_PASSWORD=your_esun_account_password   # headless（Docker）必設，否則登入會卡在互動輸入
ESUN_CONFIG_INI=esun_sdk/config.ini
# 金鑰庫解鎖密碼（cryptfile backend）。headless/Docker 必填，且**只有你知道**：
#   KEYRING_CRYPTFILE_PASSWORD=$(openssl rand -hex 16)
KEYRING_CRYPTFILE_PASSWORD=your_keyring_unlock_password
PYTHON_KEYRING_BACKEND=keyrings.cryptfile.cryptfile.CryptFileKeyring

# 選股工具要從 VM 抓回 holdings.json 時用（--sync-holdings-from-vm）
SYNC_VM_NAME=my-vm
SYNC_VM_ZONE=asia-east1-b
SYNC_VM_HOME=~/tw-selfselect
ESUN_PLACE_ORDER_TIMEOUT=25     # 單筆下單等待成交的秒數上限（5 分鐘窗口內要塞多筆 ⇒ 別設太長）
ESUN_DEBUG=0

# 凱基（BROKER=kgi 時使用）
KGI_API_KEY=your_kgi_api_key
KGI_API_SECRET=your_kgi_api_secret

# ── 3. 資料來源 ──────────────────────────────────────────────────────────
FINMIND_API_TOKEN=your_finmind_api_token_here   # 選股／回測資料；免費層有配額，跑大量回測前先看配額
# 選股候選池大小（市值前 N 大）
STOCK_NO=100
MIN_DAILY_AMOUNT=0              # 日均成交額門檻（萬元）；0=不啟用
SELECTOR_LOOKBACK_DAYS=250      # 選股只需近期資料（動能 63 日 + MA60 足夠）

# ── 4. 通知（可全部留空＝不通知）────────────────────────────────────────
TELEGRAM_BOT_TOKEN=您的Telegram機器人Token
TELEGRAM_CHAT_ID=您的ChatID
# 系統告警用另一顆 bot（選擇性；未設則沿用上面那組）
SYS_TELEGRAM_BOT_TOKEN=
SYS_TELEGRAM_CHAT_ID=
LINE_NOTIFY_TOKEN=              # LINE Notify（已於 2025 停止服務；留空即可）

# ── 5. 自選股腿（本專案的核心）──────────────────────────────────────────
# 每月自動選股：D1（每月第 1 個交易日 13:36 起）跑選股並改寫 `.env`，
# D2（第 2 個交易日 13:25–13:30）自動買進新名單。
SELFSELECT_AUTO=1               # 1=啟用在容器內自動選股（預設 0＝關，先手動跑過再開）
SELFSELECT_RISK=normal          # normal=正常（波動平衡）｜high_profit=高獲利（高風險）
SELFSELECT_TOP_N=5              # 總檔數上限 N（已持倉一律保留，只補 N−已持倉 檔新名）
SELFSELECT_LEG_PCT=100          # 自選股腿佔 TOTAL_CAPITAL 的比例 ⇒ 每檔 alloc = 本值 ÷ N
SELFSELECT_EXEC_MODE=immediate   # immediate=A（選股後就進場）｜signal=B（等 auto 進場訊號才買）
SELFSELECT_MAX_ATTEMPTS=3       # 選股失敗的重試次數上限
SELFSELECT_STATE_FILE=logs/selfselect_auto.json
MONTHLY_PICK_FILE=logs/monthly_pick.json
DOTENV_PATH=.env

# ── 6. 過熱閘門（買入當下拒買）──────────────────────────────────────────
ENTRY_OVERHEAT=1                # 1=啟用；買入當下 RSI 超買且貼近 N 日高 → 跳過該訊號
OH_RSI_TH=80                    # RSI(14) 超過此值視為超買
OH_DIST_WIN=63                  # 「距 N 日高點」的視窗（交易日）
OH_DIST_TH=8                    # 距 N 日高 ≤ 此百分比視為貼近高點
OH_ENTRY_MODE=both              # both=兩者都成立才拒買｜any=任一成立就拒買

# 選股時的「人工降溫」（達標就剔除該檔候選）
MAX_PCT_FROM_HIGH=7             # 距 52 週高 ≤ 此% ⇒ 偏熱
MAX_YTD_GAIN=100                # YTD 漲幅 > 此% ⇒ 偏熱
MAX_PER=45                      # 本益比 > 此值 ⇒ 偏貴
MIN_OVERHEAT=2                  # 上面三項命中幾項才算過熱（1=任一／2=預設／3=全中）

# ── 7. 風險控管 ─────────────────────────────────────────────────────────
MARKET_TREND_FILTER=true        # 大盤（TAIEX）在年線（MA200）之下 → 不買
MIN_DRAW_BACK=30                # 帳戶總回撤超過此% ⇒ 本次選股不換股（續抱，最多延長一次）；0=停用
BUY_AMOUNT_OFFSET=0.02          # 「已足額」的容差（避免價格跳一檔就補 1 股）
MAX_DAILY_TRADES=10             # 每日總交易次數上限
MAX_DAILY_TRADES_PER_SYMBOL=1   # 單一標的每日交易次數上限
MAX_DAILY_LOSS=0.05             # 單日虧損上限（佔 TOTAL_CAPITAL）
MAX_RISK_PER_TRADE=0.01         # 單筆風險上限（佔 TOTAL_CAPITAL）
PROFIT_MARGIN=100               # 預估損益低於此金額就不賣（避免零股 churn）

# 收盤前下單窗口（日 K 策略只在這裡判定與下單，與回測「收盤成交」同口徑）
CLOSE_ORDER_FROM=13:25

# ── 8. 執行環境 ─────────────────────────────────────────────────────────
NET_TIMEOUT_SEC=30              # 所有 HTTP 呼叫的預設 timeout（避免半開連線卡死主迴圈）
STALL_EXIT_SEC=600              # 主迴圈無進展超過此秒數 → 強制結束（交給容器 restart）
SIGNAL_DEBUG=0                  # 1=印出訊號計算細節（除錯用）

# 休市日曆：config/taiwan_holidays.json，主程式每天 08:40 檢查是否超過 30 天沒更新。

# ── 9. 個股名單（唯一真相＝PC_<代號>）───────────────────────────────────
# 每檔一行 JSON：strategy=策略名｜alloc=佔 TOTAL_CAPITAL %｜max_entry_price=進場上限價
# strategy 可選：auto（型態自動分派，建議）／bollinger／ma_cross／vwap／breakout／keep_wait
# 自選股自動選股只會增刪這個區段內的 PC_ 條目（keep_wait 條目視為人工管理，不會被動到）。
PC_2330={"strategy":"auto","alloc":20,"max_entry_price":980}
PC_2317={"strategy":"auto","alloc":20,"max_entry_price":210}
PC_2884={"strategy":"auto","alloc":20}
PC_5876={"strategy":"auto","alloc":20}
PC_2006={"strategy":"auto","alloc":20}
# 想讓某一檔用固定策略（而不是型態自動分派）就寫死 strategy：
# PC_2454={"strategy":"vwap","alloc":10}
# 想讓主程式自動偵測是否過熱才進場，省略 max_entry_price 即可。
