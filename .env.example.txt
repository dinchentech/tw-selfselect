# ── 1. 資金 ─────────────────────────────────────────────────────────────
TOTAL_CAPITAL=500000            # 可動用總資金（NT$）。每檔買入上限 = 該檔 alloc% × 本值

# ── 2. 券商 ─────────────────────────────────────────────────────────────
# esun = 玉山證券（行情＋下單）；kgi = 凱基（mock/實單）
BROKER=esun

# ── 3. 資料來源 ──────────────────────────────────────────────────────────

# ── 5. 自選股腿（本專案的核心）──────────────────────────────────────────
# 每月自動選股：D1（每月第 1 個交易日 13:36 起）跑選股並改寫 `.env`，
# D2（第 2 個交易日 13:25–13:30）自動買進新名單。
SELFSELECT_AUTO=1               # 1=啟用在容器內自動選股（預設 0＝關，先手動跑過再開）
SELFSELECT_RISK=normal          # normal=正常（波動平衡）｜high_profit=高風險（高波動；名稱是策略池特性，不代表報酬較高）
SELFSELECT_TOP_N=5              # 總檔數上限 N（已持倉一律保留，只補 N−已持倉 檔新名）
SELFSELECT_POOL_N=150           # 候選池大小（誠實池：當時市值前 N）。優先於 SELFSELECT_PRESET 的池。2026-10-29 重測（池100＋zscore＋停泊0050）：
                                #   無營收因子 +411.1%（夏普 0.75）｜有營收因子 +709.6%（夏普 0.92、MDD −53.0%）；0050 +477.6%
UNIVERSE_DELISTED=1             # 1=宇宙併入 2016-2025 已下市個股 62 檔（補倖存者偏差）；0=舊口徑（會高估大池績效）
# 選股口徑 preset（2026-10-29；實盤與回測同一個概念，回測用 PRESET=）：
#   current    = 池100、無營收因子、無產業上限（2026-10-29 之前的行為）  +411.1%／MDD −52.4%／夏普 0.75
#   revenue    = 池100＋營收因子                                        +762.4%／MDD −53.7%／夏普 0.95
#   aggressive = 池150＋營收因子＋產業上限2                            +1323.6%／MDD −40.2%／夏普 1.15  ← **預設**（0050 +477.6%／−34.0%／1.06）
# ⚠️ 後果：回撤 −40~−54%（0050 −34%）；aggressive 是 16 組實驗裡事後挑出的最好一組、沒有 walk-forward 驗證；
#          結果對資料覆蓋率極敏感（月營收 515→645 檔使同一格 +1070.5%→+1323.6%）。詳見 使用手冊 §3-0c。
SELFSELECT_PRESET=aggressive
SELFSELECT_LEG_PCT=100          # 自選股腿佔 TOTAL_CAPITAL 的比例 ⇒ 每檔 alloc = 本值 ÷ N
SELFSELECT_EXEC_MODE=immediate   # immediate=A（選股後就進場）｜signal=B（等 auto 進場訊號才買）
SELFSELECT_MAX_ATTEMPTS=3       # 選股失敗的重試次數上限
# 選不滿 N 檔時的政策（2026-10-29）：
#   proportional = 就選出的那幾檔，資金平均分配（預設）
#   topup        = 補滿：先放寬「過熱／法人」補，仍不足再依 63 日動能補
# 評分模型（2026-10-29）：zscore＝AI 代理那套固定公式（**預設**）；legacy＝舊的 ssg.score_stock 綜合分
#   （先篩 0<PER<=40，再 0.40·z(63日)+0.25·z(20日)+0.20·z(距52週高)+0.15·z(-PER)）
#   實測 2016-2025：zscore +106.9%（停泊 +369.4%）vs legacy +57.0%（停泊 +203.5%）
SELFSELECT_SCORING_MODEL=zscore
# 以下兩個是「改進實驗」用的旋鈕（2026-10-29）。目前實測結論：兩者都還沒證明有效，預設關閉。
#   FACTOR_REVENUE=1：把月營收 YoY 加進 z 分數（同池大小皆為正貢獻：池50 +291.1%→+441.7%、
#     池100 +411.1%→+762.4%、池150 +541.7%→+940.9%）
#   MAX_PER_SECTOR=2：每產業最多 2 檔（池 150 有效：+940.9%→+1323.6%；池 20 反而略差）
#   ⚠️ 這兩個是「回測旋鈕」；**實盤請用下面的 SELFSELECT_PRESET 一次切**（不要在這裡開，避免實盤行為被默默改掉）
# FACTOR_REVENUE=1
# MAX_PER_SECTOR=2
SELFSELECT_FILL_MODE=proportional
# 註：補滿（topup）只用「個股」補；ETF 只用在閒置資金停泊（IDLE_CASH_ASSET），不參與選股。
SELFSELECT_REBAL_MONTHS=1       # 換股頻率：每 N 個月選股一次（1=每月，預設＝舊行為）。12=每年 1 月選一次；非選股月份維持原持股不被換掉（策略訊號照常進出）
# AI 選股模式（可選）：1=D1 不自動選股，改由你在 DSH 貼日期完成；期限前沒完成 ⇒ D1 fallback 自動選股。
# 用法與限制見 使用手冊 §5-2b、docs/AI選股-DSH操作.md
SELFSELECT_AI=0
SELFSELECT_AI_DEADLINE=13:50
SELFSELECT_STATE_FILE=logs/selfselect_auto.json
MONTHLY_PICK_FILE=logs/monthly_pick.json
DOTENV_PATH=.env

# ── 6. 過熱閘門（買入當下拒買）──────────────────────────────────────────
ENTRY_OVERHEAT=1                # 1=啟用；買入當下 RSI 超買且貼近 N 日高 → 跳過該訊號
# 選股端過熱過濾（2026-10-31 起**預設開啟**）：距 52 週高≤MAX_PCT_FROM_HIGH%、YTD>MAX_YTD_GAIN%、
#   PER>MAX_PER 命中 ≥MIN_OVERHEAT 項就從「選股名單」剔除（與回測 Z_OVERHEAT 同一組開關）
SELFSELECT_Z_OVERHEAT=1         # 0=關（回到 2026-10-31 之前的行為：zscore 完全不降溫）
OH_RSI_TH=80                    # RSI(14) 超過此值視為超買
OH_DIST_WIN=63                  # 「距 N 日高點」的視窗（交易日）
OH_DIST_TH=8                    # 距 N 日高 ≤ 此百分比視為貼近高點
OH_ENTRY_MODE=both              # both=兩者都成立才拒買｜any=任一成立就拒買

# 選股時的「人工降溫」（達標就剔除該檔候選）
MAX_PCT_FROM_HIGH=7             # 距 52 週高 ≤ 此% ⇒ 偏熱
MAX_YTD_GAIN=100                # YTD 漲幅 > 此% ⇒ 偏熱
MAX_PER=45                      # 本益比 > 此值 ⇒ 偏貴
MIN_OVERHEAT=2                  # 上面三項命中幾項才算過熱（1=任一／2=預設／3=全中）

# ── 7. 風險控管 ─────────────────────────────────────────────────────────
MARKET_TREND_FILTER=true        # 大盤（TAIEX）在年線（MA200）之下 → 不買
MIN_DRAW_BACK=30                # 帳戶總回撤超過此% ⇒ 本次選股不換股（續抱，最多延長一次）；0=停用
BUY_AMOUNT_OFFSET=0.02          # 「已足額」的容差（避免價格跳一檔就補 1 股）
MAX_DAILY_TRADES=10             # 每日總交易次數上限
MAX_DAILY_TRADES_PER_SYMBOL=1   # 單一標的每日交易次數上限
MAX_DAILY_LOSS=0.05             # 單日虧損上限（佔 TOTAL_CAPITAL）
MAX_RISK_PER_TRADE=0.01         # 單筆風險上限（佔 TOTAL_CAPITAL）
PROFIT_MARGIN=100               # 預估損益低於此金額就不賣（避免零股 churn）

# 收盤前下單窗口（日 K 策略只在這裡判定與下單，與回測「收盤成交」同口徑）
CLOSE_ORDER_FROM=13:25
